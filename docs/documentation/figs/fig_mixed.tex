\begin{tikzpicture}[fig]
\newcommand{\cloud}[4]{% centre x, y, colour, seed
  \pgfmathsetseed{#4}
  \foreach \n in {1,...,6} \fill[#3] ({#1+0.12*(rand+rand)},{#2+0.12*(rand+rand)}) circle (1.6pt);}
\newcommand{\ring}[4]{%
  \pgfmathsetseed{#4}
  \foreach \n in {1,...,6} \draw[#3, thick] ({#1+0.12*(rand+rand)},{#2+0.12*(rand+rand)}) circle (1.6pt);}
% ordinary
\begin{scope}
  \draw[black!30] (0,0) rectangle (2.8,2.0);
  \cloud{1.4}{1.0}{cP0}{3} \ring{1.4}{1.0}{cP0}{8}
  \node[lab] at (1.4,2.3) {ordinary point};
  \node[small, text width=2.9cm, anchor=north] at (1.4,-0.1) {halves $a_1$ (dots) and $a_2$ (rings) mixed: $E_a\approx0$};
\end{scope}
% mixed
\begin{scope}[xshift=3.6cm]
  \draw[black!30] (0,0) rectangle (2.8,2.0);

  \node[lab] at (1.4,2.3) {mixed point};
  \node[small, text width=2.9cm, anchor=north] at (1.4,-0.1) {runs end in two states, in different shares in the two halves: $E_a$ large};
